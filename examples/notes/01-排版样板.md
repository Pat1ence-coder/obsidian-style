$$
\begin{aligned}
\mathcal{L}(\theta)
&= -\log p(y_{1:n}\mid x_{1:n},\theta) \\
&= -\sum_{i=1}^{n}\log p(y_i\mid x_i,\theta) \\
\theta_{t+1}
&= \theta_t-\eta\,\frac{\widehat{m}_t}{\sqrt{\widehat{v}_t}+\varepsilon}.
